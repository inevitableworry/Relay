cmake_minimum_required(VERSION 3.20)
project(RelaySocialLink VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RELAY_PS5 "Cross-compile with PS5 payload SDK" OFF)
add_executable(relay-social-link source/main.cpp vendor/qrcodegen/qrcodegen.cpp)
target_include_directories(relay-social-link PRIVATE vendor vendor/qrcodegen)
if(RELAY_PS5)
  if(NOT DEFINED ENV{PS5_PAYLOAD_SDK})
    message(FATAL_ERROR "Set PS5_PAYLOAD_SDK")
  endif()
  set(SDK "$ENV{PS5_PAYLOAD_SDK}")
  set(HB "${SDK}/target/user/homebrew")
  target_include_directories(relay-social-link PRIVATE
    "${HB}/include" "${HB}/include/SDL2")
  execute_process(COMMAND "${SDK}/bin/prospero-pkg-config" --libs --static
    sdl2 SDL2_ttf libcurl
    RESULT_VARIABLE PKG_STATUS OUTPUT_VARIABLE PKG_LINK
    OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT PKG_STATUS EQUAL 0)
    message(FATAL_ERROR "PS5 SDL2_ttf/libcurl dependencies missing")
  endif()
  separate_arguments(PKG_LIBS UNIX_COMMAND "${PKG_LINK}")
  target_link_libraries(relay-social-link PRIVATE
    ${PKG_LIBS} kernel SceUserService m pthread)
  target_compile_definitions(relay-social-link PRIVATE RELAY_PS5=1)
  target_compile_options(relay-social-link PRIVATE -O2 -fPIC -femulated-tls)
  set_target_properties(relay-social-link PROPERTIES SUFFIX ".elf")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf libcurl)
  find_package(Threads REQUIRED)
  target_link_libraries(relay-social-link PRIVATE PkgConfig::DEPS Threads::Threads)
endif()
